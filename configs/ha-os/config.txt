# /boot/firmware/config.txt - Autohome Hub (RPi 5 + 7" DSI Touchscreen + HifiBerry DAC+)

# Display settings
hdmi_force_hotplug=1
hdmi_group=2
hdmi_mode=82

# 7" Official DSI Touchscreen
dtoverlay=rpi-7inch-touchscreen,rotate=90

# HifiBerry DAC+ HAT
dtoverlay=hifiberry-dacplus

# Audio settings
dtparam=audio=off
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# GPU memory (RPi 5 manages this automatically; keep minimal for DSI)
gpu_mem=64

# Disable onboard audio to avoid conflicts with HifiBerry
dtparam=audio=off

# I2C for sensors
dtparam=i2c_arm=on
dtparam=i2c1=on

# SPI (for potential future use)
dtparam=spi=on

# UART for serial console (if needed)
enable_uart=1

# Watchdog
dtparam=watchdog=on

# Reduce boot time
boot_delay=0
